> def _tree(s):
>     """A reading (lawread's string) as nested (op, [args]); leaves as strings."""
>     s = s.strip()
>     i = s.find("(")
>     if i < 0:
>         return s
>     body, args, depth, start = s[i + 1:-1], [], 0, 0
>     for j, ch in enumerate(body):
>         if ch == "(":
>             depth += 1
>         elif ch == ")":
>             depth -= 1
>         elif ch == "," and depth == 0:
>             args.append(_tree(body[start:j]))
>             start = j + 1
>     args.append(_tree(body[start:]))
>     return (s[:i], args)
